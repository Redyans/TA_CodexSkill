`session_end.py`：

```python
import json
import os
import re
import shutil
import sys
from datetime import datetime
from pathlib import Path


def text_of(value):
    if isinstance(value, str):
        return value
    if isinstance(value, list):
        return "\n".join(x for x in (text_of(v) for v in value) if x)
    if isinstance(value, dict):
        for key in ("text", "input_text", "output_text", "content"):
            if key in value:
                result = text_of(value[key])
                if result:
                    return result
    return ""


def find_messages(value):
    if isinstance(value, dict):
        role = value.get("role")
        if role in ("user", "assistant"):
            content = text_of(value.get("content", value.get("text", ""))).strip()
            if content:
                yield role, content

        for child in value.values():
            yield from find_messages(child)

    elif isinstance(value, list):
        for child in value:
            yield from find_messages(child)


def main():
    event = json.load(sys.stdin)

    codex_home = Path(os.environ.get("CODEX_HOME", Path.home() / ".codex"))
    output_dir = codex_home / "memories" / "sessions"
    output_dir.mkdir(parents=True, exist_ok=True)

    session_id = str(event.get("session_id", "unknown"))
    safe_id = re.sub(r"[^A-Za-z0-9_.-]", "_", session_id)
    stamp = datetime.now().astimezone().strftime("%Y-%m-%d_%H-%M-%S")

    markdown_path = output_dir / f"{stamp}_{safe_id}.md"
    transcript_path = event.get("transcript_path")

    lines = [
        "# Codex 会话记忆",
        "",
        f"- session_id: `{session_id}`",
        f"- ended_at: `{datetime.now().astimezone().isoformat()}`",
        f"- cwd: `{event.get('cwd', '')}`",
        "",
    ]

    if transcript_path and Path(transcript_path).exists():
        seen = set()

        for role, content in find_messages(
            Path(transcript_path).read_text(encoding="utf-8", errors="replace").splitlines()
        ):
            key = (role, content)
            if key in seen:
                continue
            seen.add(key)

            title = "用户" if role == "user" else "Codex"
            lines.extend([f"## {title}", "", content, ""])

        # 同时保留原始 JSONL，便于以后重新解析
        shutil.copyfile(transcript_path, markdown_path.with_suffix(".jsonl"))

    else:
        lines.extend(["没有找到 transcript 文件。", ""])

    markdown_path.write_text("\n".join(lines), encoding="utf-8")


if __name__ == "__main__":
    main()
```
